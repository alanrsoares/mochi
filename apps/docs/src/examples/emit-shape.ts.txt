export type Shape =
  | { _tag: "Circle"; _0: number }
  | { _tag: "Rect"; _0: number; _1: number };

import { _curry, pi, square } from "@mochi/runtime";

const Circle = (_0: number): Shape => ({ _tag: "Circle", _0 });
const area: (shape: Shape) => number = (shape: Shape) => {
  const $match = shape;
  switch ($match._tag) {
    case "Circle": {
      const { _0: r } = $match;
      return pi * square(r);
    }
    case "Rect": {
      const { _0: w, _1: h } = $match;
      return w * h;
    }
    default: {
      throw new Error("non-exhaustive match");
    }
  }
};
const widen: <A>(
  k: number,
  box: { w: number } & A,
) => { w: number } & A = _curry(
  2,
  <A>(k: number, box: { w: number } & A) => ({ ...box, w: box.w * k }),
);
const circleArea = area(Circle(2));
const wider = widen(3, { w: 2, label: "box" });
